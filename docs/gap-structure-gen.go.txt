// Command gapsvg renders the gap-value structure of every sequence surveyed in
// docs/GAP_SEQUENCES.md as a log-scale strip plot.
package main

import (
	"fmt"
	"math"
	"os"
	"sort"
	"strings"
)

type seq struct {
	gaps  func(n int) []int
	name  string
	group string // "practical" | "pratt" | "historical"
	note  string
}

func asc(d []int) []int { sort.Ints(d); return d }

func shell(n int) []int {
	var out []int
	for h := n / 2; h >= 1; h /= 2 {
		out = append(out, h)
	}
	return asc(out)
}

func frankLazarus(n int) []int {
	var out []int
	for k := 1; ; k++ {
		h := 2*(n/(1<<(k+1))) + 1
		if h <= 1 {
			break
		}
		out = append(out, h)
	}
	out = append(out, 1)
	return asc(out)
}

func hibbard(n int) []int {
	var out []int
	for k := 1; ; k++ {
		h := 1<<k - 1
		if h >= n {
			break
		}
		out = append(out, h)
	}
	return asc(out)
}

func papernov(n int) []int {
	out := []int{1}
	for k := 1; ; k++ {
		h := 1<<k + 1
		if h >= n {
			break
		}
		out = append(out, h)
	}
	return asc(out)
}

// smooth returns every a^p * b^q below n.
func smooth(a, b int) func(n int) []int {
	return func(n int) []int {
		var out []int
		for p := 1; p < n; p *= a {
			for q := p; q < n; q *= b {
				out = append(out, q)
			}
		}
		return asc(out)
	}
}

func knuth(n int) []int {
	out := []int{}
	for h := 1; h < n/3+1; h = 3*h + 1 {
		out = append(out, h)
	}
	if len(out) == 0 {
		out = []int{1}
	}
	return asc(out)
}

func incerpiSedgewick(n int) []int {
	table := []int{1, 3, 7, 21, 48, 112, 336, 861, 1968, 4592, 13776, 33936, 86961, 198768}
	var out []int
	for _, h := range table {
		if h < n {
			out = append(out, h)
		}
	}
	return asc(out)
}

func sedgewick82(n int) []int {
	out := []int{1}
	for k := 1; ; k++ {
		h := 1<<(2*k) + 3*(1<<(k-1)) + 1
		if h >= n {
			break
		}
		out = append(out, h)
	}
	return asc(out)
}

func sedgewick86(n int) []int {
	var out []int
	for i := 0; ; i++ {
		var h float64
		if i%2 == 0 {
			h = 9*math.Pow(2, float64(i)) - 9*math.Pow(2, float64(i)/2) + 1
		} else {
			h = 8*math.Pow(2, float64(i)) - 6*math.Pow(2, float64(i+1)/2) + 1
		}
		if int(h) >= n {
			break
		}
		out = append(out, int(h))
	}
	return asc(out)
}

func gonnetBaezaYates(n int) []int {
	var out []int
	h := n
	for {
		h = (5*h - 1) / 11
		if h < 1 {
			h = 1
		}
		out = append(out, h)
		if h == 1 {
			break
		}
	}
	return asc(out)
}

func tokuda(n int) []int {
	var out []int
	hp := 1.0
	for {
		h := int(math.Ceil(hp))
		if h >= n {
			break
		}
		out = append(out, h)
		hp = 2.25*hp + 1
	}
	return asc(out)
}

// ciuraFrom extends a fixed table by the conventional 2.25 ratio.
func ciuraFrom(table []int) func(n int) []int {
	return func(n int) []int {
		var out []int
		for _, h := range table {
			if h < n {
				out = append(out, h)
			}
		}
		if len(out) == 0 {
			return []int{1}
		}
		for h := int(2.25 * float64(out[len(out)-1])); h < n; h = int(2.25 * float64(h)) {
			out = append(out, h)
		}
		return asc(out)
	}
}

func lee(n int) []int {
	const g = 2.243609061420001
	var out []int
	for k := 1; ; k++ {
		h := int(math.Ceil((math.Pow(g, float64(k)) - 1) / (g - 1)))
		if h >= n {
			break
		}
		out = append(out, h)
	}
	return asc(out)
}

func skeanB(n int) []int {
	const a, b, c, d = 4.0816, 8.5714, 2.2449, 0.0
	out := []int{1}
	for i := 0; ; i++ {
		h := int(a*math.Pow(b, float64(i)/c) + d)
		if h >= n {
			break
		}
		out = append(out, h)
	}
	return asc(out)
}

var sequences = []seq{
	{name: "Tokuda 1992", group: "practical", note: "recommended default", gaps: tokuda},
	{name: "Ciura 2001 (Large)", group: "practical", note: "best measured comparisons", gaps: ciuraFrom([]int{1, 4, 10, 23, 57, 132, 301, 701, 1750})},
	{name: "Ciura-1000", group: "practical", note: "searched for N=1000", gaps: ciuraFrom([]int{1, 4, 10, 23, 57, 156, 409, 995})},
	{name: "Ciura-128", group: "practical", note: "searched for N=128", gaps: ciuraFrom([]int{1, 4, 9, 24, 85, 126})},
	{name: "Lee 2021", group: "practical", note: "tuned Tokuda variant", gaps: lee},
	{name: "Skean 2023 (Ours-B)", group: "practical", note: "grid-searched, N=10000", gaps: skeanB},
	{name: "Knuth 1973", group: "practical", note: "integer-only fallback", gaps: knuth},
	{name: "Gonnet-Baeza-Yates 1991", group: "historical", note: "N-derived, ratio 2.2", gaps: gonnetBaezaYates},
	{name: "Sedgewick 1986", group: "historical", note: "current repo default", gaps: sedgewick86},
	{name: "Sedgewick 1982", group: "historical", note: "ratio 4", gaps: sedgewick82},
	{name: "Incerpi-Sedgewick 1985", group: "historical", note: "best asymptotic exponent", gaps: incerpiSedgewick},
	{name: "Hibbard 1963", group: "historical", note: "repo alternative", gaps: hibbard},
	{name: "Papernov-Stasevich 1965", group: "historical", note: "ratio 2", gaps: papernov},
	{name: "Pratt 1971 (2^p 3^q)", group: "pratt", note: "best worst case", gaps: smooth(2, 3)},
	{name: "Pratt-25 (2^p 5^q)", group: "pratt", note: "fewest exchanges", gaps: smooth(2, 5)},
	{name: "Pratt-34 (3^p 4^q)", group: "pratt", note: "fewest comparisons of family", gaps: smooth(3, 4)},
	{name: "Frank-Lazarus 1960", group: "historical", note: "N-derived, odd-forced", gaps: frankLazarus},
	{name: "Shell 1959", group: "historical", note: "negative control, N^2", gaps: shell},
}

const (
	nMax   = 100000
	labelW = 190.0
	ratioX = labelW + 26
	countX = labelW + 74
	plotX0 = labelW + 116
	plotW  = 428.0
	rowH   = 25.0
	topPad = 108.0
	svgW   = plotX0 + plotW + 24
	dotR   = 3.0
)

func xPos(v int) float64 {
	if v < 1 {
		v = 1
	}
	return plotX0 + math.Log10(float64(v))/math.Log10(nMax)*plotW
}

// ratio reports the median multiplicative step, a robust stand-in for the
// asymptotic growth rate of an irregular sequence.
func ratio(d []int) float64 {
	if len(d) < 3 {
		return 0
	}
	var rs []float64
	for i := 1; i < len(d); i++ {
		rs = append(rs, float64(d[i])/float64(d[i-1]))
	}
	sort.Float64s(rs)
	return rs[len(rs)/2]
}

func main() {
	var b strings.Builder
	svgH := topPad + rowH*float64(len(sequences)) + 58

	fmt.Fprintf(&b, `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 %.0f %.0f" width="%.0f" height="%.0f" font-family="system-ui, -apple-system, &quot;Segoe UI&quot;, sans-serif">`, svgW, svgH, svgW, svgH)

	b.WriteString(`
<style>
  .surface { fill: #fcfcfb; }
  .ink     { fill: #0b0b0b; }
  .ink2    { fill: #52514e; }
  .muted   { fill: #898781; }
  .grid    { stroke: #e1e0d9; }
  .axis    { stroke: #c3c2b7; }
  .dot-practical  { fill: #2a78d6; stroke: #fcfcfb; }
  .dot-pratt      { fill: #eb6834; stroke: #fcfcfb; }
  .dot-historical { fill: #898781; stroke: #fcfcfb; }
  .rule    { stroke: #e1e0d9; }
  @media (prefers-color-scheme: dark) {
    .surface { fill: #1a1a19; }
    .ink     { fill: #ffffff; }
    .ink2    { fill: #c3c2b7; }
    .grid    { stroke: #2c2c2a; }
    .axis    { stroke: #383835; }
    .rule    { stroke: #2c2c2a; }
    .dot-practical  { fill: #3987e5; stroke: #1a1a19; }
    .dot-pratt      { fill: #d95926; stroke: #1a1a19; }
    .dot-historical { fill: #898781; stroke: #1a1a19; }
  }
</style>
`)
	fmt.Fprintf(&b, `<rect class="surface" width="%.0f" height="%.0f"/>`, svgW, svgH)

	// Title block.
	fmt.Fprintf(&b, `<text class="ink" x="18" y="30" font-size="15" font-weight="600">Shellsort gap sequences: where the gaps actually land</text>`)
	fmt.Fprintf(&b, `<text class="ink2" x="18" y="50" font-size="11.5">Every gap below N = 100,000, on a log scale. One dot per gap. Spacing shows growth ratio; density shows pass count.</text>`)

	// Legend.
	legend := []struct{ cls, label string }{
		{"dot-practical", "practical candidates"},
		{"dot-pratt", "Pratt family"},
		{"dot-historical", "historical / controls"},
	}
	lx := 18.0
	for _, l := range legend {
		fmt.Fprintf(&b, `<circle class="%s" cx="%.1f" cy="70" r="%.1f" stroke-width="1.2"/>`, l.cls, lx+4, dotR)
		fmt.Fprintf(&b, `<text class="ink2" x="%.1f" y="74" font-size="11">%s</text>`, lx+14, l.label)
		lx += 14 + float64(len(l.label))*5.9 + 24
	}

	// Axis gridlines + ticks.
	yTop := topPad - 16
	yBot := topPad + rowH*float64(len(sequences)) - 6
	ticks := []struct {
		v     int
		label string
	}{{1, "1"}, {10, "10"}, {100, "100"}, {1000, "1,000"}, {10000, "10,000"}, {100000, "100,000"}}
	for _, t := range ticks {
		x := xPos(t.v)
		fmt.Fprintf(&b, `<line class="grid" x1="%.1f" y1="%.1f" x2="%.1f" y2="%.1f" stroke-width="1"/>`, x, yTop, x, yBot)
		fmt.Fprintf(&b, `<text class="muted" x="%.1f" y="%.1f" font-size="10" text-anchor="middle">%s</text>`, x, yTop-6, t.label)
	}
	fmt.Fprintf(&b, `<text class="muted" x="%.1f" y="%.1f" font-size="10" text-anchor="middle">gap value</text>`, plotX0+plotW/2, yBot+30)

	// Column headers.
	fmt.Fprintf(&b, `<text class="muted" x="%.1f" y="%.1f" font-size="10" text-anchor="middle">ratio</text>`, ratioX+12, yTop-6)
	fmt.Fprintf(&b, `<text class="muted" x="%.1f" y="%.1f" font-size="10" text-anchor="middle">passes</text>`, countX+14, yTop-6)

	for i, s := range sequences {
		y := topPad + rowH*float64(i)
		d := s.gaps(nMax)

		if i > 0 {
			fmt.Fprintf(&b, `<line class="rule" x1="18" y1="%.1f" x2="%.1f" y2="%.1f" stroke-width="1"/>`, y-9, svgW-18, y-9)
		}
		fmt.Fprintf(&b, `<text class="ink" x="18" y="%.1f" font-size="11.5">%s</text>`, y+2, s.name)
		fmt.Fprintf(&b, `<text class="muted" x="18" y="%.1f" font-size="9">%s</text>`, y+12, s.note)

		for _, h := range d {
			fmt.Fprintf(&b, `<circle class="dot-%s" cx="%.2f" cy="%.1f" r="%.1f" stroke-width="1.2"/>`, s.group, xPos(h), y, dotR)
		}

		r := ratio(d)
		rs := "varies"
		if r > 0 {
			rs = fmt.Sprintf("%.2f", r)
		}
		fmt.Fprintf(&b, `<text class="ink2" x="%.1f" y="%.1f" font-size="10.5" text-anchor="middle" font-variant-numeric="tabular-nums">%s</text>`, ratioX+12, y+3, rs)
		fmt.Fprintf(&b, `<text class="ink2" x="%.1f" y="%.1f" font-size="10.5" text-anchor="middle" font-variant-numeric="tabular-nums">%d</text>`, countX+14, y+3, len(d))
	}

	b.WriteString(`</svg>`)

	if err := os.WriteFile(os.Args[1], []byte(b.String()), 0o644); err != nil {
		panic(err)
	}

	// Console check of the first terms, so the plot can be trusted.
	for _, s := range sequences {
		d := s.gaps(nMax)
		n := 8
		if len(d) < n {
			n = len(d)
		}
		fmt.Printf("%-26s %v  (count %d, ratio %.3f)\n", s.name, d[:n], len(d), ratio(d))
	}
}
